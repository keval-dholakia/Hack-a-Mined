  order_no character varying NOT NULL UNIQUE,
  transporter_id bigint NOT NULL,
  so_id bigint,
  pickup_date date,
  destination character varying,
  vehicle_type character varying,
  vehicle_no character varying,
  driver_name character varying,
  driver_mobile character varying,
  status character varying DEFAULT 'Booked'::character varying CHECK (status::text = ANY (ARRAY['Booked'::character varying, 'In Transit'::character varying, 'Delivered'::character varying]::text[])),
  remarks text,
  created_at timestamp without time zone DEFAULT now(),
  updated_at timestamp without time zone DEFAULT now(),
  CONSTRAINT transport_orders_pkey PRIMARY KEY (id),
  CONSTRAINT transport_orders_so_id_fkey FOREIGN KEY (so_id) REFERENCES public.sale_orders(id),
  CONSTRAINT transport_orders_transporter_id_fkey FOREIGN KEY (transporter_id) REFERENCES public.transport_masters(id)
);
CREATE TABLE public.users (
  id bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  name character varying NOT NULL,
  email character varying NOT NULL UNIQUE,
  password character varying NOT NULL,
  role_id bigint NOT NULL,
  is_active smallint DEFAULT 1,
  last_login_at timestamp without time zone,
  created_at timestamp without time zone DEFAULT now(),
  updated_at timestamp without time zone DEFAULT now(),
  CONSTRAINT users_pkey PRIMARY KEY (id),
  CONSTRAINT users_role_id_fkey FOREIGN KEY (role_id) REFERENCES public.roles(id)
);
CREATE TABLE public.vendors (
  id bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  code character varying UNIQUE,
  name character varying NOT NULL,
  gstin character varying,
  pan character varying,
  contact_person character varying,
  mobile character varying,
  email character varying,
  address text,
  city character varying,
  state character varying,
  pincode character varying,
  bank_name character varying,
  bank_account character varying,
  bank_ifsc character varying,
  payment_terms integer DEFAULT 30,
  is_active smallint DEFAULT 1,
  created_at timestamp without time zone DEFAULT now(),
  updated_at timestamp without time zone DEFAULT now(),
  CONSTRAINT vendors_pkey PRIMARY KEY (id)
);
CREATE TABLE public.vouchers (
  id bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  voucher_no character varying NOT NULL UNIQUE,
  voucher_type character varying NOT NULL CHECK (voucher_type::text = ANY (ARRAY['Journal'::character varying, 'Payment'::character varying, 'Receipt'::character varying, 'Contra'::character varying]::text[])),
  voucher_date date NOT NULL DEFAULT CURRENT_DATE,
  party_name character varying,
  debit_account character varying,
  credit_account character varying,
  amount numeric NOT NULL,
  mode character varying DEFAULT 'NEFT'::character varying CHECK (mode::text = ANY (ARRAY['Cash'::character varying, 'Cheque'::character varying, 'NEFT'::character varying, 'UPI'::character varying]::text[])),
  bank_account character varying,
  ref_no character varying,
  narration text,
  created_at timestamp without time zone DEFAULT now(),
  CONSTRAINT vouchers_pkey PRIMARY KEY (id)
);
CREATE TABLE public.warehouses (
  id bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  code character varying UNIQUE,
  name character varying NOT NULL,
  address text,
  city character varying,
  state character varying,
  manager_name character varying,
  manager_mobile character varying,
  is_active smallint DEFAULT 1,
  created_at timestamp without time zone DEFAULT now(),
  updated_at timestamp without time zone DEFAULT now(),
  CONSTRAINT warehouses_pkey PRIMARY KEY (id)
);
